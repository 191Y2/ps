    14→    14→    const d = new Date(ts), now = new Date();
    15→    15→    const days = Math.floor((new Date(now.toDateString()) - new Date(d.toDateString())) / 86400000);
    16→    16→    if (days <= 0) return "Today";
    17→    17→    if (days === 1) return "Yesterday";
    18→    18→    if (days <= 7) return "Previous 7 days";
    19→    19→    if (days <= 30) return "Previous 30 days";
    20→    20→    return "Older";
    21→    21→  }

   275→   275→  function renderContent(m, streaming) {
   276→   276→    const nodes = [];
   277→   277→    const segs = Ace.MD.split(m.content || "");
   278→   278→    segs.forEach((seg, idx) => {
   279→   279→      if (seg.kind === "md") {
   280→   280→        if (!seg.text.trim()) return;
   281→   281→        const div = Ace.el("div", "md");
   282→   282→        div.innerHTML = Ace.MD.render(seg.text);
   283→   283→        nodes.push(div);
   284→   284→      } else { // fence
   285→   285→        const looksUI = Ace.Blocks.isBlock(seg.lang) || seg.lang === "json" || seg.lang === "";
   286→   286→        if (looksUI && !seg.closed && m.streaming) {
   287→   287→          nodes.push(Ace.el("div", "blk blk-skeleton", `<div class="skel-line"></div><div class="skel-line short"></div><div class="skel-line"></div><div class="skel-note">${Ace.ICON("spark", 13)} composing UI…</div>`));
   288→   288→        } else if (looksUI) {
   289→   289→          nodes.push(Ace.Blocks.render(seg.lang, seg.body, blockCtx(m, idx)));
   290→   290→        } else if (seg.lang === "html" && m.artifacts && m.artifacts.length) {
   291→   291→          // handled by work.js trace/artifact UI; skip raw echo
   292→   292→        } else if (seg.body && seg.body.trim()) {
   293→   293→          nodes.push(Ace.Blocks.renderCode(seg.lang, seg.body));
   294→   294→        }
   295→   295→      }
   296→   296→    });
   297→   297→    return nodes;
   298→   298→  }

   300→   300→  function blockCtx(m, idx) {
   301→   301→    return {
   302→   302→      messageId: m.id, blockIndex: idx,
   303→   303→      state: (m.uiStates || {})[idx],
   304→   304→      save: (patch) => {
   305→   305→        m.uiStates = m.uiStates || {};
   306→   306→        m.uiStates[idx] = Object.assign({}, m.uiStates[idx], patch);
   307→   307→        C.persistUIStates(m);
   308→   308→      }
   309→   309→    };
   310→   310→  }

   316→   316→    }, 900);

   353→   353→  C.send = async () => {
   354→   354→    if (C.sending) return;
   355→   355→    const input = $("input");
   356→   356→    const text = input.value.trim();
   357→   357→    const images = Ace.Images.pending.slice();
   358→   358→    if (!text && !images.length) return;
   359→   359→    if (!Ace.API.hasKey()) { Ace.UI.openSettings(); return; }
   360→   360→
   361→   361→    input.value = ""; Ace.App.resizeInput();
   362→   362→    Ace.Images.pending.length = 0; Ace.Images.renderChips();
   363→   363→
   364→   364→    const mode = Ace.App.mode;
   365→   365→    if (mode === "work") return Ace.Work.send(text, images); // work mode pipeline
   366→   366→
   367→   367→    const chat = await ensureChat(mode);
   368→   368→    const userMsg = { id: Ace.uid(), role: "user", content: text, images, createdAt: Date.now() };
   369→   369→    C.messages.push(userMsg);
   370→   370→    C.appendMessage(userMsg);
   371→   371→    await Ace.Store.addMessage(chat.id, userMsg);
   372→   372→    if (!chat.title) {
   373→   373→      chat.title = text.slice(0, 40) || "Images";
   374→   374→      Ace.Store.updateChat(chat.id, { title: chat.title }).catch(() => {});
   375→   375→      Ace.Memory.genTitle(text || "image analysis", chat.id).then(t => {
   376→   376→        chat.title = t; C.renderSidebar();
   377→   377→      }).catch(() => {});
   378→   378→    }

   380→   380→    await runCompletion(chat);
   381→   381→  };

   383→   383→  async function runCompletion(chat, retryOf) {
   384→   384→    C.sending = true;
   385→   385→    Ace.App.setSending(true);
   386→   386→    const ai = retryOf || {
   387→   387→      id: Ace.uid(), role: "assistant", content: "", createdAt: Date.now(),
   388→   388→      model: "", flags: { think: Ace.App.toggles.think, web: Ace.App.toggles.web, deep: Ace.App.toggles.deep },
   389→   389→      uiStates: {}
   390→   390→    };
   391→   391→    if (!retryOf) { C.messages.push(ai); }
   392→   392→    C.rerenderMessage(ai);
   393→   393→    C.scrollBottom();
   394→   394→
   395→   395→    const lastUser = [...C.messages].reverse().find(m => m.role === "user");
   396→   396→    const userText = (lastUser && lastUser.content) || "";
   397→   397→    const hasImages = !!(lastUser && lastUser.images && lastUser.images.length);
   398→   398→
   399→   399→    try {
   400→   400→      // context assembly
   401→   401→      const settings = await Ace.Store.getSettings();
   402→   402→      const memories = await Ace.Memory.recall(userText).catch(() => []);
   403→   403→      const memHint = $("memHint"); if (memHint) memHint.textContent = memories.length ? `🧠 ${memories.length} memories in context` : "";
   404→   404→
   405→   405→      const autoWeb = settings.autoWebSearch !== false && !hasImages && !Ace.App.toggles.web &&
   406→   406→        /\b(latest|today|right now|currently|news|prices?)\b|\bwho won\b|\bweather\b|\b(up ?to ?date|update on)\b|\b(stock|election|released|coming out|game last night)\b|20(2[5-9]|3[0-9])/i.test(userText);
   407→   407→      const useWeb = Ace.App.toggles.web || autoWeb;
   408→   408→      const useDeep = Ace.App.toggles.deep && !hasImages;
   409→   409→      if (autoWeb) Ace.UI.toast("Auto web search on — this looks like a current-events question", "info");
   410→   410→
   411→   411→      if (useDeep) {
   412→   412→        await runDeep(chat, ai, userText);
   413→   413→      } else {
   414→   414→        // free live references (Wikipedia + DDG) — real citations even on free keys
   415→   415→        let refs = [];
   416→   416→        if (useWeb) {
   417→   417→          refs = await Ace.API.freeWebSearch(userText).catch(() => []);
   418→   418→        }
   419→   419→        const sysPrompt = Ace.PROMPTS.chat({
   420→   420→          webSearch: useWeb, refs, memories, summary: chat.summary,
   421→   421→          quizHint: /quiz|test me|flash ?card|drill me|practice (test|questions?)|study for/i.test(userText),
   422→   422→          imageCount: hasImages ? lastUser.images.length : 0
   423→   423→        });
   424→   424→        const history = C.messages.slice(-Ace.HISTORY_LIMIT)
   425→   425→          .map(m => ({
   426→   426→            role: m.role, content: m.content,
   427→   427→            // images only survive on the LAST user message (vision routing);
   428→   428→            // older image messages degrade to text so text models don't reject them
   429→   429→            images: (m.role === "user" && m === lastUser) ? m.images : undefined
   430→   430→          }));
   431→   431→        history.unshift({ role: "system", content: sysPrompt });
   432→   432→
   433→   433→        let model, chain, tools = null, thinking = undefined;
   434→   434→        if (hasImages) { model = Ace.ROUTE.vision; chain = Ace.FALLBACKS.vision; }
   435→   435→        else if (useWeb) { model = Ace.ROUTE.web; chain = [Ace.ROUTE.web]; tools = Ace.API.WEB_TOOLS; thinking = "disabled"; }
   436→   436→        else if (Ace.App.toggles.think) { model = Ace.ROUTE.think; chain = Ace.FALLBACKS.text; }
   437→   437→        else { model = Ace.ROUTE.chat; chain = [Ace.ROUTE.chat]; }
   438→   438→
   439→   439→        const streamOpts = {
   440→   440→          model, fallbackChain: chain, taskKey: "chat", tools, thinking,
   441→   441→          messages: history, signal: (C.abort = new AbortController()).signal
   442→   442→        };
   443→   443→        if (tools) { streamOpts.temperature = 0.5; streamOpts.maxTokens = 4000; }
   444→   444→
   445→   445→        const onDelta = (content, reasoning) => {
   446→   446→          ai.content = content; ai.reasoning = reasoning; ai.streaming = true;
   447→   447→          C.rerenderMessage(ai, true);
   448→   448→          C.scrollBottom();
   449→   449→        };
   450→   450→        const res = tools
   451→   451→          ? await Ace.API.searchChat(Object.assign(streamOpts, { onDelta }))
   452→   452→          : await Ace.API.stream(Object.assign(streamOpts, { onDelta }));
   453→   453→
   454→   454→        ai.content = res.content; ai.reasoning = res.reasoning || "";
   455→   455→        ai.model = res.model; ai.fallbackUsed = res.fallbackUsed;
   456→   456→        ai.streaming = false;
   457→   457→        const toolSources = (res.sources && res.sources.length) ? res.sources : null;
   458→   458→        ai.searchResult = toolSources || (useWeb && refs.length ? refs : null);
   459→   459→        C.rerenderMessage(ai);
   460→   460→        await Ace.Store.addMessage(chat.id, ai);
   461→   461→      }
   462→   462→
   463→   463→      // background: memory + summary
   464→   464→      Ace.Memory.extract(C.messages.slice(0, -1), chat.id).then(r => {
   465→   465→        if (r && (r.added || r.updated || r.deleted)) {
   466→   466→          const parts = [];
   467→   467→          if (r.added) parts.push(`${r.added} new`);
   468→   468→          if (r.updated) parts.push(`${r.updated} updated`);
   469→   469→          Ace.UI.toast(`🧠 Memory: ${parts.join(", ")}`, "ok");
   470→   470→        }
   471→   471→      }).catch(() => {});
   472→   472→      if (C.messages.length >= Ace.SUMMARIZE_AFTER && C.messages.length % 8 === 0) {
   473→   473→        Ace.Memory.summarize(chat, C.messages).catch(() => {});
   474→   474→      }
   475→   475→      await Ace.Store.updateChat(chat.id, { msgCount: C.messages.length, updatedAt: Date.now() });
   476→   476→    } catch (e) {
   477→   477→      if (e.name === "AbortError") {
   478→   478→        ai.streaming = false; ai.content = ai.content || "*(stopped)*";
   479→   479→        C.rerenderMessage(ai);
   480→   480→        if (ai.content && ai.content !== "*(stopped)*") await Ace.Store.addMessage(chat.id, ai);
   481→   481→      } else {
   482→   482→        ai.streaming = false; ai.error = true;
   483→   483→        ai.content = `**Something went wrong.** ${Ace.esc(Ace.API.friendlyError(e))}`;
   484→   484→        C.rerenderMessage(ai);
   485→   485→      }
   486→   486→    } finally {
   487→   487→      C.sending = false; C.abort = null;
   488→   488→      Ace.App.setSending(false);
   489→   489→      Ace.Images.pending.length = 0;
   490→   490→    }
   491→   491→  }

   493→   493→  /* ---- deep research pipeline ---- */
   494→   494→  async function runDeep(chat, ai) {
   495→   495→    const lastUser = [...C.messages].reverse().find(m => m.role === "user");
   496→   496→    const query = (lastUser && lastUser.content) || "";
   497→   497→    ai.flags.deep = true; ai.flags.web = true;
   498→   498→    ai.deepSteps = [];
   499→   499→    const traceEl = () => {
   500→   500→      const prev = document.querySelector(`[data-mid="${ai.id}"] .deep-steps`);
   501→   501→      if (prev) prev.remove();
   502→   502→      const box = Ace.el("div", "deep-steps");
   503→   503→      ai.deepSteps.forEach(s => box.append(Ace.el("div", "deep-step " + (s.state === "run" ? "run" : "done"),
   504→   504→        `${s.state === "run" ? '<span class="spin"></span>' : Ace.ICON("check", 13)} ${Ace.esc(s.label)}`)));
   505→   505→      const msgEl = document.querySelector(`[data-mid="${ai.id}"] .msg-head`);
   506→   506→      if (msgEl && msgEl.nextSibling) msgEl.parentNode.insertBefore(box, msgEl.nextSibling);
   507→   507→      C.scrollBottom();
   508→   508→    };
   509→   509→    const res = await Ace.API.deepResearch({
   510→   510→      query, chain: Ace.FALLBACKS.text, signal: (C.abort = new AbortController()).signal,
   511→   511→      onStep: (label, state) => {
   512→   512→        const ex = ai.deepSteps.find(s => s.label.startsWith(label.split(" —")[0]));
   513→   513→        if (ex) { ex.label = label; ex.state = state; } else ai.deepSteps.push({ label, state });
   514→   514→        traceEl();
   515→   515→      }
   516→   516→    });
   517→   517→    ai.content = res.report; ai.model = Ace.ROUTE.deep;
   518→   518→    ai.searchResult = res.sources; ai.streaming = false;
   519→   519→    ai.reasoning = "";
   520→   520→    C.rerenderMessage(ai);
   521→   521→    await Ace.Store.addMessage(chat.id, ai);
   522→   522→  }
