    18→    const d = new Date(ts), now = new Date();
    19→    const days = Math.floor((new Date(now.toDateString()) - new Date(d.toDateString())) / 86400000);
    20→    if (days <= 0) return "Today";
    21→    if (days === 1) return "Yesterday";
    22→    if (days <= 7) return "Previous 7 days";
    23→    if (days <= 30) return "Previous 30 days";
    24→    return "Older";

   163→  function renderTaskBar(m) {
   164→    const tasks = m.tasks || [];
   165→    if (!tasks.length) return null;
   166→    const running = m.streaming !== false && tasks.some(t => !t.t1);
   167→    const bar = Ace.el("div", "taskbar");
   168→    tasks.forEach((t, i) => {
   169→      const live = !t.t1;
   170→      const ms = live ? Date.now() - t.t0 : t.t1 - t.t0;
   171→      const chip = Ace.el("span", "task-chip" + (live ? " run" : " done"),
   172→        `${live ? '<span class="task-dot"></span>' : Ace.ICON("check", 11)}<span class="task-lbl">${Ace.esc(t.label)}</span><span class="task-time">${fmtSecs(ms)}s</span>`);
   173→      chip.dataset.taskIdx = i;
   174→      bar.append(chip);
   175→    });
   176→    const last = tasks[tasks.length - 1];
   177→    if (!running && last && last.t1) {
   178→      const total = tasks.reduce((a, t) => a + ((t.t1 || t.t0) - t.t0), 0);
   179→      const tot = Ace.el("span", "task-total", `${Ace.ICON("clock", 11)} Worked for ${fmtSecs(total)}s`);
   180→      bar.append(tot);
   181→    }
   182→    return bar;
   183→  }

   339→  /* ---- content segments: markdown runs + UI blocks + code fences ---- */
   340→  function renderContent(m, streaming) {
   341→    const nodes = [];
   342→    const segs = Ace.MD.split(m.content || "");
   343→    segs.forEach((seg, idx) => {
   344→      if (seg.kind === "md") {
   345→        if (!seg.text.trim()) return;
   346→        const div = Ace.el("div", "md");
   347→        div.innerHTML = Ace.MD.render(seg.text);
   348→        nodes.push(div);
   349→      } else { // fence
   350→        const looksUI = Ace.Blocks.isBlock(seg.lang) || seg.lang === "json" || seg.lang === "";
   351→        if (looksUI && !seg.closed && m.streaming) {
   352→          nodes.push(Ace.el("div", "blk blk-skeleton", `<div class="skel-line"></div><div class="skel-line short"></div><div class="skel-line"></div><div class="skel-note">${Ace.ICON("spark", 13)} composing UI…</div>`));
   353→        } else if (looksUI) {
   354→          nodes.push(Ace.Blocks.render(seg.lang, seg.body, blockCtx(m, idx)));
   355→        } else if (seg.lang === "html" && m.artifacts && m.artifacts.length) {
   356→          // handled by work.js trace/artifact UI; skip raw echo
   357→        } else if (seg.body && seg.body.trim()) {
   358→          nodes.push(Ace.Blocks.renderCode(seg.lang, seg.body));
   359→        }
   360→      }
   361→    });
   362→    return nodes;
   363→  }

   405→    if (C.activeChat) return C.activeChat;
   406→    const chat = {
   407→      id: Ace.uid(), title: "", mode: mode || "chat",
   408→      createdAt: Date.now(), updatedAt: Date.now(), pinned: false, msgCount: 0, summary: ""

   417→  /* ================= SEND FLOW ================= */
   418→  C.send = async () => {
   419→    if (C.sending) return;
   420→    const input = $("input");
   421→    const text = input.value.trim();
   422→    const images = Ace.Images.pending.slice();
   423→    const files = Ace.Files ? Ace.Files.take() : [];
   424→    if (!text && !images.length && !files.length) return;
   425→    if (!Ace.API.hasKey()) { Ace.UI.openSettings(); return; }
   426→
   427→    input.value = ""; Ace.App.resizeInput();
   428→    Ace.Images.pending.length = 0; Ace.Images.renderChips();
   429→
   430→    const mode = Ace.App.mode;
   431→    if (mode === "work") return Ace.Work.send(text, images, files); // agent mode pipeline
   432→
   433→    const chat = await ensureChat(mode);
   434→    const userMsg = { id: Ace.uid(), role: "user", content: text, images, files, createdAt: Date.now() };
   435→    C.messages.push(userMsg);
   436→    C.appendMessage(userMsg);
   437→    await Ace.Store.addMessage(chat.id, userMsg);
   438→    if (!chat.title) {
   439→      chat.title = text.slice(0, 40) || (images.length ? "Images" : "Files");
   440→      Ace.Store.updateChat(chat.id, { title: chat.title }).catch(() => {});
   441→      Ace.Memory.genTitle(text || "image analysis", chat.id).then(t => {
   442→        chat.title = t; C.renderSidebar();
   443→      }).catch(() => {});
   444→    }
   445→    C.renderSidebar();
   446→    await runCompletion(chat);
   447→  };

   449→  /* task helpers bound to an assistant message */
   450→  function taskBegin(ai, k, label) {
   451→    ai.tasks = ai.tasks || [];
   452→    const ex = ai.tasks.find(t => t.k === k);
   453→    if (ex) { if (ex.t1) { ex.t1 = null; ex.label = label; } }
   454→    else ai.tasks.push({ k, label, t0: Date.now(), t1: null });
   455→    if (ai.streaming !== false) startTaskTicker(ai);
   456→    C.rerenderMessage(ai);
   457→  }
   458→  function taskEnd(ai, k) {
   459→    const t = (ai.tasks || []).find(t => t.k === k && !t.t1);
   460→    if (t) t.t1 = Date.now();
   461→    C.rerenderMessage(ai);
   462→  }
   463→  C.taskBegin = taskBegin;   // used by work.js (agentic trace timings)
   464→  C.taskEnd = taskEnd;
   465→  C.finishTasks = finishTasks;

   467→  async function runCompletion(chat, retryOf) {
   468→    C.sending = true;
   469→    Ace.App.setSending(true);
   470→    const t0 = Date.now();
   471→    const ai = retryOf || {
   472→      id: Ace.uid(), role: "assistant", content: "", createdAt: Date.now(),
   473→      model: "", flags: { think: Ace.App.toggles.think, web: Ace.App.toggles.web, deep: Ace.App.toggles.deep },
   474→      uiStates: {}, tasks: []
   475→    };
   476→    ai.tasks = ai.tasks || [];
   477→    if (!retryOf) { C.messages.push(ai); }
   478→    ai.streaming = true;
   479→    C.rerenderMessage(ai);
   480→    C.scrollBottom();
   481→    startTaskTicker(ai);
   482→
   483→    const lastUser = [...C.messages].reverse().find(m => m.role === "user");
   484→    const userText = (lastUser && lastUser.content) || "";
   485→    const hasImages = !!(lastUser && lastUser.images && lastUser.images.length);
   486→    const hasFiles = !!(lastUser && lastUser.files && lastUser.files.length);
   487→
   488→    try {
   489→      // context assembly
   490→      const settings = await Ace.Store.getSettings();
   491→      const memories = await Ace.Memory.recall(userText).catch(() => []);
   492→      const memHint = $("memHint"); if (memHint) memHint.textContent = memories.length ? `${memories.length} memories in context` : "";
   493→
   494→      const autoWeb = settings.autoWebSearch !== false && !hasImages && !hasFiles && !Ace.App.toggles.web &&
   495→        /\b(latest|today|right now|current|currently|news|prices?|record|score|champion)\b|\bwho won\b|\bweather\b|\b(up ?to ?date|update on)\b|\b(stock|election|released|coming out|game last night)\b|20(2[5-9]|3[0-9])/i.test(userText);
   496→      const useWeb = Ace.App.toggles.web || autoWeb;
   497→      const useDeep = Ace.App.toggles.deep && !hasImages;
   498→      if (autoWeb) Ace.UI.toast("Auto web search on — this looks like a current-events question", "info");
   499→
   500→      if (useDeep) {
   501→        await runDeep(chat, ai, userText);
   502→      } else {
   503→        /* ---- REAL web search (blocks the reply; stopwatch shows live) ---- */
   504→        let refs = [];
   505→        if (useWeb && !hasImages) {
   506→          taskBegin(ai, "search", "Searching the web");
   507→          refs = await Ace.API.searchWeb(userText).catch(() => []);
   508→          taskEnd(ai, "search");
   509→          if (!refs.length && Ace.App.toggles.web)
   510→            Ace.UI.toast("The web search came back empty — Ace will answer from what it knows and say so", "warn");
   511→        }
   512→        ai.flags.web = useWeb && !!refs.length; // honest flag: only claim it if it actually happened
   513→
   514→        const sysPrompt = Ace.PROMPTS.chat({
   515→          webSearch: useWeb, refs, memories, summary: chat.summary,
   516→          quizHint: /quiz|test me|flash ?card|drill me|practice (test|questions?)|study for/i.test(userText),
   517→          imageCount: hasImages ? lastUser.images.length : 0,
   518→          fileCount: hasFiles ? lastUser.files.length : 0
   519→        });
   520→        const history = C.messages.slice(-Ace.HISTORY_LIMIT)
   521→          .map(m => {
   522→            const isLastUser = m.role === "user" && m === lastUser;
   523→            let content = m.content;
   524→            // files ride along with the LAST user message (full contents for the AI to view)
   525→            if (isLastUser && m.files && m.files.length && Ace.Files)
   526→              content = (content || "") + Ace.Files.buildContextFrom(m.files);
   527→            return {
   528→              role: m.role, content,
   529→              // images only survive on the LAST user message (vision routing);
   530→              // older image messages degrade to text so text models don't reject them
   531→              images: (m.role === "user" && m === lastUser) ? m.images : undefined
   532→            };
   533→          });
   534→        history.unshift({ role: "system", content: sysPrompt });
   535→
   536→        /* routing: free-first. Think = reasoning ON (same free model).
   537→           Images route to the vision model. Everything else runs with
   538→           thinking OFF for instant first tokens. */
   539→        let model, chain, thinking = "disabled";
   540→        if (hasImages) { model = Ace.ROUTE.vision; chain = Ace.chainFor(Ace.ROUTE.vision, "vision"); thinking = "disabled"; }
   541→        else if (Ace.App.toggles.think) { model = Ace.ROUTE.think; chain = Ace.chainFor(Ace.ROUTE.think, "text"); thinking = "enabled"; }
   542→        else { model = Ace.ROUTE.chat; chain = Ace.chainFor(Ace.ROUTE.chat, "text"); }
   543→
   544→        let streamDone = false; // ignore any late delta racing after completion
   545→        const onDelta = (content, reasoning) => {
   546→          if (streamDone) return;
   547→          ai.content = content; ai.reasoning = reasoning; ai.streaming = true;
   548→          // phase detection for the stopwatch
   549→          if (reasoning && !content) taskBegin(ai, "think", "Thinking");
   550→          else if (content) {
   551→            if ((ai.tasks || []).some(t => t.k === "think" && !t.t1)) taskEnd(ai, "think");
   552→            const opens = (content.match(/```/g) || []).length;
   553→            if (opens % 2 === 1) { // we're inside an open fence right now
   554→              const fm = content.slice(content.lastIndexOf("```")).match(/^```(\w+)/);
   555→              if (fm && Ace.Blocks.isBlock(fm[1]))
   556→                taskBegin(ai, "ui", "Creating " + (/quiz/.test(fm[1]) ? "quiz" : fm[1]) + " UI");
   557→              else taskBegin(ai, "ui", "Writing code");
   558→            } else {
   559→              taskBegin(ai, "write", "Writing");
   560→            }
   561→          }
   562→          C.rerenderMessage(ai, true);
   563→          C.scrollBottom();
   564→        };
   565→        const res = await Ace.API.stream({
   566→          model, fallbackChain: chain, taskKey: "chat", thinking,
   567→          messages: history, signal: (C.abort = new AbortController()).signal,
   568→          onDelta
   569→        });
   570→        streamDone = true;
   571→
   572→        ai.content = res.content; ai.reasoning = res.reasoning || "";
   573→        ai.model = res.model; ai.fallbackUsed = res.fallbackUsed;
   574→        ai.streaming = false;
   575→        ai.searchResult = (useWeb && refs.length) ? refs : null;
   576→        finishTasks(ai);
   577→        C.rerenderMessage(ai);
   578→        await Ace.Store.addMessage(chat.id, ai);
   579→      }
   580→
   581→      // background: memory + summary (never blocks the reply)
   582→      Ace.Memory.extract(C.messages.slice(0, -1), chat.id).then(r => {
   583→        if (r && (r.added || r.updated || r.deleted)) {
   584→          const parts = [];
   585→          if (r.added) parts.push(`${r.added} new`);
   586→          if (r.updated) parts.push(`${r.updated} updated`);
   587→          Ace.UI.toast(`Memory: ${parts.join(", ")}`, "ok");
   588→        }
   589→      }).catch(() => {});
   590→      if (C.messages.length >= Ace.SUMMARIZE_AFTER && C.messages.length % 8 === 0) {
   591→        Ace.Memory.summarize(chat, C.messages).catch(() => {});
   592→      }
   593→      await Ace.Store.updateChat(chat.id, { msgCount: C.messages.length, updatedAt: Date.now() });
   594→    } catch (e) {
   595→      finishTasks(ai);
   596→      if (e.name === "AbortError") {
   597→        ai.streaming = false; ai.content = ai.content || "*(stopped)*";
   598→        C.rerenderMessage(ai);
   599→        if (ai.content && ai.content !== "*(stopped)*") await Ace.Store.addMessage(chat.id, ai);
   600→      } else {
   601→        ai.streaming = false; ai.error = true;
   602→        ai.content = `**Something went wrong.** ${Ace.esc(Ace.API.friendlyError(e))}`;
   603→        C.rerenderMessage(ai);
   604→      }
   605→    } finally {
   606→      C.sending = false; C.abort = null;
   607→      Ace.App.setSending(false);
   608→      Ace.Images.pending.length = 0;
   609→    }
   610→  }

   612→  /* ---- deep research pipeline ---- */
   613→  async function runDeep(chat, ai) {
   614→    const lastUser = [...C.messages].reverse().find(m => m.role === "user");
   615→    const query = (lastUser && lastUser.content) || "";
   616→    ai.flags.deep = true;
   617→    ai.flags.web = true;
   618→    const res = await Ace.API.deepResearch({
   619→      query, chain: Ace.chainFor(Ace.ROUTE.deep, "text"),
   620→      signal: (C.abort = new AbortController()).signal,
   621→      onStep: (label, state) => {
   622→        const k = "step-" + label.split(" ")[0].toLowerCase();
   623→        if (state === "run") taskBegin(ai, k, label);
   624→        else taskEnd(ai, k);
   625→      }
   626→    });
   627→    ai.content = res.report; ai.model = res.model || Ace.ROUTE.deep;
   628→    ai.searchResult = res.sources; ai.streaming = false;
   629→    ai.reasoning = "";
   630→    finishTasks(ai);
   631→    C.rerenderMessage(ai);
   632→    await Ace.Store.addMessage(chat.id, ai);
   633→  }
