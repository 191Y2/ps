   163→   163→  function renderTaskBar(m) {
   164→   164→    const tasks = m.tasks || [];
   165→   165→    if (!tasks.length) return null;
   166→   166→    const running = m.streaming !== false && tasks.some(t => !t.t1);
   167→   167→    const bar = Ace.el("div", "taskbar");
   168→   168→    tasks.forEach((t, i) => {
   169→   169→      const live = !t.t1;
   170→   170→      const ms = live ? Date.now() - t.t0 : t.t1 - t.t0;
   171→   171→      const chip = Ace.el("span", "task-chip" + (live ? " run" : " done"),
   172→   172→        `${live ? '<span class="task-dot"></span>' : Ace.ICON("check", 11)}<span class="task-lbl">${Ace.esc(t.label)}</span><span class="task-time">${fmtSecs(ms)}s</span>`);
   173→   173→      chip.dataset.taskIdx = i;
   174→   174→      bar.append(chip);
   175→   175→    });
   176→   176→    const last = tasks[tasks.length - 1];
   177→   177→    if (!running && last && last.t1) {
   178→   178→      const total = tasks.reduce((a, t) => a + ((t.t1 || t.t0) - t.t0), 0);
   179→   179→      const tot = Ace.el("span", "task-total", `${Ace.ICON("clock", 11)} Worked for ${fmtSecs(total)}s`);
   180→   180→      bar.append(tot);
   181→   181→    }
   182→   182→    return bar;
   183→   183→  }

   329→   329→  function renderContent(m, streaming) {
   330→   330→    const nodes = [];
   331→   331→    const segs = Ace.MD.split(m.content || "");
   332→   332→    segs.forEach((seg, idx) => {
   333→   333→      if (seg.kind === "md") {
   334→   334→        if (!seg.text.trim()) return;
   335→   335→        const div = Ace.el("div", "md");
   336→   336→        div.innerHTML = Ace.MD.render(seg.text);
   337→   337→        nodes.push(div);
   338→   338→      } else { // fence
   339→   339→        const looksUI = Ace.Blocks.isBlock(seg.lang) || seg.lang === "json" || seg.lang === "";
   340→   340→        if (looksUI && !seg.closed && m.streaming) {
   341→   341→          nodes.push(Ace.el("div", "blk blk-skeleton", `<div class="skel-line"></div><div class="skel-line short"></div><div class="skel-line"></div><div class="skel-note">${Ace.ICON("spark", 13)} composing UI…</div>`));
   342→   342→        } else if (looksUI) {
   343→   343→          nodes.push(Ace.Blocks.render(seg.lang, seg.body, blockCtx(m, idx)));
   344→   344→        } else if (seg.lang === "html" && m.artifacts && m.artifacts.length) {
   345→   345→          // handled by work.js trace/artifact UI; skip raw echo
   346→   346→        } else if (seg.body && seg.body.trim()) {
   347→   347→          nodes.push(Ace.Blocks.renderCode(seg.lang, seg.body));
   348→   348→        }
   349→   349→      }
   350→   350→    });
   351→   351→    return nodes;
   352→   352→  }

   354→   354→  function blockCtx(m, idx) {
   355→   355→    return {
   356→   356→      messageId: m.id, blockIndex: idx,
   357→   357→      state: (m.uiStates || {})[idx],
   358→   358→      save: (patch) => {
   359→   359→        m.uiStates = m.uiStates || {};
   360→   360→        m.uiStates[idx] = Object.assign({}, m.uiStates[idx], patch);
   361→   361→        C.persistUIStates(m);
   362→   362→      }
   363→   363→    };
   364→   364→  }
   365→   365→  const uiTimers = {};
   366→   366→  C.persistUIStates = (m) => {
   367→   367→    clearTimeout(uiTimers[m.id]);
   368→   368→    uiTimers[m.id] = setTimeout(() => {
   369→   369→      if (C.activeChat) Ace.Store.updateMessage(C.activeChat.id, m.id, { uiStates: m.uiStates || {} }).catch(() => {});
   370→   370→    }, 900);
   371→   371→  };

   455→   455→  async function runCompletion(chat, retryOf) {
   456→   456→    C.sending = true;
   457→   457→    Ace.App.setSending(true);
   458→   458→    const t0 = Date.now();
   459→   459→    const ai = retryOf || {
   460→   460→      id: Ace.uid(), role: "assistant", content: "", createdAt: Date.now(),
   461→   461→      model: "", flags: { think: Ace.App.toggles.think, web: Ace.App.toggles.web, deep: Ace.App.toggles.deep },
   462→   462→      uiStates: {}, tasks: []
   463→   463→    };
   464→   464→    ai.tasks = ai.tasks || [];
   465→   465→    if (!retryOf) { C.messages.push(ai); }
   466→   466→    ai.streaming = true;
   467→   467→    C.rerenderMessage(ai);
   468→   468→    C.scrollBottom();
   469→   469→    startTaskTicker(ai);
   470→   470→
   471→   471→    const lastUser = [...C.messages].reverse().find(m => m.role === "user");
   472→   472→    const userText = (lastUser && lastUser.content) || "";
   473→   473→    const hasImages = !!(lastUser && lastUser.images && lastUser.images.length);
   474→   474→
   475→   475→    try {
   476→   476→      // context assembly
   477→   477→      const settings = await Ace.Store.getSettings();
   478→   478→      const memories = await Ace.Memory.recall(userText).catch(() => []);
   479→   479→      const memHint = $("memHint"); if (memHint) memHint.textContent = memories.length ? `${memories.length} memories in context` : "";
   480→   480→
   481→   481→      const autoWeb = settings.autoWebSearch !== false && !hasImages && !Ace.App.toggles.web &&
   482→   482→        /\b(latest|today|right now|currently|news|prices?)\b|\bwho won\b|\bweather\b|\b(up ?to ?date|update on)\b|\b(stock|election|released|coming out|game last night)\b|20(2[5-9]|3[0-9])/i.test(userText);
   483→   483→      const useWeb = Ace.App.toggles.web || autoWeb;
   484→   484→      const useDeep = Ace.App.toggles.deep && !hasImages;
   485→   485→      if (autoWeb) Ace.UI.toast("Auto web search on — this looks like a current-events question", "info");
   486→   486→
   487→   487→      if (useDeep) {
   488→   488→        await runDeep(chat, ai, userText);
   489→   489→      } else {
   490→   490→        /* ---- REAL web search (blocks the reply; stopwatch shows live) ---- */
   491→   491→        let refs = [];
   492→   492→        if (useWeb && !hasImages) {
   493→   493→          taskBegin(ai, "search", "Searching the web");
   494→   494→          refs = await Ace.API.searchWeb(userText).catch(() => []);
   495→   495→          taskEnd(ai, "search");
   496→   496→          if (!refs.length && Ace.App.toggles.web)
   497→   497→            Ace.UI.toast("The web search came back empty — Ace will answer from what it knows and say so", "warn");
   498→   498→        }
   499→   499→        ai.flags.web = useWeb && !!refs.length; // honest flag: only claim it if it actually happened
   500→   500→
   501→   501→        const sysPrompt = Ace.PROMPTS.chat({
   502→   502→          webSearch: useWeb, refs, memories, summary: chat.summary,
   503→   503→          quizHint: /quiz|test me|flash ?card|drill me|practice (test|questions?)|study for/i.test(userText),
   504→   504→          imageCount: hasImages ? lastUser.images.length : 0
   505→   505→        });
   506→   506→        const history = C.messages.slice(-Ace.HISTORY_LIMIT)
   507→   507→          .map(m => ({
   508→   508→            role: m.role, content: m.content,
   509→   509→            // images only survive on the LAST user message (vision routing);
   510→   510→            // older image messages degrade to text so text models don't reject them
   511→   511→            images: (m.role === "user" && m === lastUser) ? m.images : undefined
   512→   512→          }));
   513→   513→        history.unshift({ role: "system", content: sysPrompt });
   514→   514→
   515→   515→        /* routing: free-first. Think = reasoning ON (same free model).
   516→   516→           Images route to the vision model. Everything else runs with
   517→   517→           thinking OFF for instant first tokens. */
   518→   518→        let model, chain, thinking = "disabled";
   519→   519→        if (hasImages) { model = Ace.ROUTE.vision; chain = Ace.chainFor(Ace.ROUTE.vision, "vision"); thinking = "disabled"; }
   520→   520→        else if (Ace.App.toggles.think) { model = Ace.ROUTE.think; chain = Ace.chainFor(Ace.ROUTE.think, "text"); thinking = "enabled"; }
   521→   521→        else { model = Ace.ROUTE.chat; chain = Ace.chainFor(Ace.ROUTE.chat, "text"); }
   522→   522→
   523→   523→        let streamDone = false; // ignore any late delta racing after completion
   524→   524→        const onDelta = (content, reasoning) => {
   525→   525→          if (streamDone) return;
   526→   526→          ai.content = content; ai.reasoning = reasoning; ai.streaming = true;
   527→   527→          // phase detection for the stopwatch
   528→   528→          if (reasoning && !content) taskBegin(ai, "think", "Thinking");
   529→   529→          else if (content) {
   530→   530→            if ((ai.tasks || []).some(t => t.k === "think" && !t.t1)) taskEnd(ai, "think");
   531→   531→            const opens = (content.match(/```/g) || []).length;
   532→   532→            if (opens % 2 === 1) { // we're inside an open fence right now
   533→   533→              const fm = content.slice(content.lastIndexOf("```")).match(/^```(\w+)/);
   534→   534→              if (fm && Ace.Blocks.isBlock(fm[1]))
   535→   535→                taskBegin(ai, "ui", "Creating " + (/quiz/.test(fm[1]) ? "quiz" : fm[1]) + " UI");
   536→   536→              else taskBegin(ai, "ui", "Writing code");
   537→   537→            } else {
   538→   538→              taskBegin(ai, "write", "Writing");
   539→   539→            }
   540→   540→          }
   541→   541→          C.rerenderMessage(ai, true);
   542→   542→          C.scrollBottom();
   543→   543→        };
   544→   544→        const res = await Ace.API.stream({
   545→   545→          model, fallbackChain: chain, taskKey: "chat", thinking,
   546→   546→          messages: history, signal: (C.abort = new AbortController()).signal,
   547→   547→          onDelta
   548→   548→        });
   549→   549→        streamDone = true;
   550→   550→
   551→   551→        ai.content = res.content; ai.reasoning = res.reasoning || "";
   552→   552→        ai.model = res.model; ai.fallbackUsed = res.fallbackUsed;
   553→   553→        ai.streaming = false;
   554→   554→        ai.searchResult = (useWeb && refs.length) ? refs : null;
   555→   555→        finishTasks(ai);
   556→   556→        C.rerenderMessage(ai);
   557→   557→        await Ace.Store.addMessage(chat.id, ai);
   558→   558→      }
   559→   559→
   560→   560→      // background: memory + summary (never blocks the reply)
   561→   561→      Ace.Memory.extract(C.messages.slice(0, -1), chat.id).then(r => {
   562→   562→        if (r && (r.added || r.updated || r.deleted)) {
   563→   563→          const parts = [];
   564→   564→          if (r.added) parts.push(`${r.added} new`);
   565→   565→          if (r.updated) parts.push(`${r.updated} updated`);
   566→   566→          Ace.UI.toast(`Memory: ${parts.join(", ")}`, "ok");
   567→   567→        }
   568→   568→      }).catch(() => {});
   569→   569→      if (C.messages.length >= Ace.SUMMARIZE_AFTER && C.messages.length % 8 === 0) {
   570→   570→        Ace.Memory.summarize(chat, C.messages).catch(() => {});
   571→   571→      }
   572→   572→      await Ace.Store.updateChat(chat.id, { msgCount: C.messages.length, updatedAt: Date.now() });
   573→   573→    } catch (e) {
   574→   574→      finishTasks(ai);
   575→   575→      if (e.name === "AbortError") {
   576→   576→        ai.streaming = false; ai.content = ai.content || "*(stopped)*";
   577→   577→        C.rerenderMessage(ai);
   578→   578→        if (ai.content && ai.content !== "*(stopped)*") await Ace.Store.addMessage(chat.id, ai);
   579→   579→      } else {
   580→   580→        ai.streaming = false; ai.error = true;
   581→   581→        ai.content = `**Something went wrong.** ${Ace.esc(Ace.API.friendlyError(e))}`;
   582→   582→        C.rerenderMessage(ai);
   583→   583→      }
   584→   584→    } finally {
   585→   585→      C.sending = false; C.abort = null;
   586→   586→      Ace.App.setSending(false);
   587→   587→      Ace.Images.pending.length = 0;
   588→   588→    }
   589→   589→  }

   609→   609→    finishTasks(ai);
